    // ===================================================================
    // ===== 比比看 班級作答紀錄（貼在 class_scores 規則之後、最後兩個 } 之前）=====
    // 依賴既有函式：signedIn() / realUser() / isAdmin() / subActive(uid)
    // 班級（teacher_classes、classes）沿用跑吧！漢字、色數消消樂的共用規則，不用改。
    // 資料：
    //   class_plays/{自動 id} → 學生每玩完一回合寫一筆
    //     { cid, owner, game:'bbk', h, nick, at, th, lv, rg, score, max, qs }
    //     h    = 暱稱雜湊（16 碼，同暱稱不分大小寫與空白視為同一人）
    //     th   = 主題包（people…）
    //     lv   = 關卡 1–3；rg = 老師範圍（'' = 一般模式，否則像 'd=height,age&t=pick&n=10'）
    //     qs   = 每題一筆 { t 題型, d 特徵, f 句型, p 方向(1/-1/0), ok 是否答對, f1 是否第一次就答對, ids 人物, tv 對錯題的句子本身是對是錯 }
    // 規則：
    //   - 學生（匿名登入）知道班級碼才能寫；老師要訂閱中才收得到
    //   - 寫入後不能修改；只有老師本人（或管理員）讀得到、刪得掉
    //   - 到期後老師仍可讀、匯出、刪除
    // ===================================================================
    match /class_plays/{pid} {
      function okPlay() {
        let d = request.resource.data;
        return d.keys().hasOnly(['cid','owner','game','h','nick','at','th','lv','rg','score','max','qs'])
          && d.keys().hasAll(['cid','owner','game','h','nick','at','th','lv','rg','score','max','qs'])
          && d.game == 'bbk'
          && d.cid is string && d.cid.matches('^[a-z0-9]{8,32}$')
          && d.owner == get(/databases/$(database)/documents/classes/$(d.cid)).data.owner
          && d.h is string && d.h.matches('^[a-f0-9]{16}$')
          && d.nick is string && d.nick.size() >= 1 && d.nick.size() <= 16
          && d.at == request.time
          && d.th is string && d.th.size() <= 20
          && d.lv is int && d.lv >= 1 && d.lv <= 3
          && d.rg is string && d.rg.size() <= 200
          && d.max is int && d.max >= 10 && d.max <= 300
          && d.score is int && d.score >= 0 && d.score <= d.max
          && d.qs is list && d.qs.size() >= 1 && d.qs.size() <= 30
          && subActive(d.owner);
      }
      allow create: if signedIn() && okPlay();
      allow get, list: if signedIn() && resource.data.owner == request.auth.uid;
      allow update: if false;
      allow delete: if realUser() && (resource.data.owner == request.auth.uid || isAdmin());
    }
